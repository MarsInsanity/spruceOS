#!/bin/bash
# GUI launch hook: which SDL2 pysdl2 loads, and library paths, per device.
command -v export_sdl_gamecontroller_map >/dev/null 2>&1 || . /mnt/SDCARD/spruce/scripts/helperFunctions.sh

# A board whose cfg names its own SDL2 set (DEVICE_PYSDL2_DLL_PATH; the MagicX boards: without
# a path pysdl2 finds no SDL2 at all and the app dies on `import sdl2`, 2026-09-22).
[ -n "$DEVICE_PYSDL2_DLL_PATH" ] && export PYSDL2_DLL_PATH="$DEVICE_PYSDL2_DLL_PATH"

case "$PLATFORM" in
    Flip|SmartProS)          export PYSDL2_DLL_PATH="/mnt/SDCARD/Persistent/portmaster/lib/" ;;
    Brick|SmartPro|BrickPro) export PYSDL2_DLL_PATH="/mnt/SDCARD/spruce/brick/sdl2" ;;
    Anbernic*)               export PYSDL2_DLL_PATH="/mnt/SDCARD/App/PyUI/dll-mali" ;;
    RGB30|Miniloong|A10Mini|A10MiniV4|XUMiniM|G350|V10|RG351P|RG353M|E5Plus|Pixel2DarkMoss|RPMiniV2) export PYSDL2_DLL_PATH="/usr/lib/aarch64-linux-gnu" ;;
esac

case "$PLATFORM" in
    RGB30|Miniloong|A10Mini|A10MiniV4|XUMiniM|G350|V10|RG351P|RG353M|E5Plus|Pixel2DarkMoss|RPMiniV2) ;;
    *) export PATH="/mnt/SDCARD/spruce/flip/bin:$PATH" ;;
esac

case "$PLATFORM" in
    Flip|SmartProS|Brick|SmartPro|BrickPro)
        export LD_LIBRARY_PATH="/mnt/SDCARD/spruce/flip/lib:$LD_LIBRARY_PATH"
        ;;
    Anbernic*)
        export LD_LIBRARY_PATH="$LD_LIBRARY_PATH:/mnt/SDCARD/spruce/h700/ports-lib64"
        export_sdl_gamecontroller_map positional
        ;;
    RGB30|Miniloong|A10Mini|A10MiniV4|XUMiniM|G350|V10|RG351P|Pixel2DarkMoss|RPMiniV2)
        export_sdl_gamecontroller_map positional
        ;;
esac

# Retroid Pocket Mini V2 (LightMoss): PortMaster's LOVE splash (utils/pmsplash.txt)
# and pugwash both use the system SDL2's KMSDRM driver; the splash starts first and
# holds DRM master, so SDL's device probe in pugwash skipped the screen ("No
# available video device") before pugwash got to stop the splash. With the probe's
# master check off, SDL takes the screen anyway and becomes master at window
# creation (KMSDRM_GBMInit), which pugwash does only after the splash has quit.
case "$PLATFORM" in
    RPMiniV2) export SDL_KMSDRM_REQUIRE_DRM_MASTER=0 ;;
esac
